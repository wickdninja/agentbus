# fish completion for agentbus, written by: agentbus completion fish
# Load it for every shell: agentbus completion fish > ~/.config/fish/completions/agentbus.fish
function __agentbus_word
    set -l tokens (commandline -opc)
    set -e tokens[1]
    set -l skip 0
    set -l seen 0
    for t in $tokens
        if test $skip -eq 1
            set skip 0
            continue
        end
        switch $t
            case --after --as --before --branch --cwd --depth --id --interval --kind --limit --model --purpose --reply-to --request-id --role --scope --since --to --tool --topic --ttl-min --up
                set skip 1
            case '-*'
            case '*'
                set seen (math $seen + 1)
                if test $seen -eq $argv[1]
                    echo $t
                    return
                end
        end
    end
end
